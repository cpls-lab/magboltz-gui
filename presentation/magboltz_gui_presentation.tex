% Slide 2 — Magboltz: Why It Matters
\begin{frame}{Magboltz in Detector Physics}
\begin{itemize}
  \item Monte Carlo electron transport simulation
  \item Fundamental tool for:
    \begin{itemize}
      \item Gas detector design
      \item Drift properties
      \item Diffusion
      \item Townsend / attachment
    \end{itemize}
  \item Used across:
    \begin{itemize}
      \item MPGDs
      \item TPCs
      \item RPC studies
      \item Simulation chains (Garfield++)
    \end{itemize}
\end{itemize}
\vspace{0.3cm}
\textbf{Magboltz remains a cornerstone of gas detector modeling.}
\end{frame}

% Slide 10 — Setup (pip + bundled Qt)
\begin{frame}[fragile]{Setup: pip (bundled Qt)}
\textbf{Recommended when you want Qt bundled with the app.}
\vspace{0.3cm}

\begin{itemize}
  \item Python environment
  \item Magboltz binary available in PATH
\end{itemize}

\textbf{Commands:}
\begin{verbatim}
pip install "magboltz-gui[qt]"
magboltz-gui
\end{verbatim}
\end{frame}

% Slide 11 — Setup (pip + system Qt)
\begin{frame}[fragile]{Setup: pip (system Qt)}
\textbf{Recommended on Linux when using system PyQt6.}
\vspace{0.3cm}

\begin{itemize}
  \item Install system Qt bindings
  \item Magboltz binary available in PATH
\end{itemize}

\textbf{Commands (Debian/Ubuntu):}
\begin{verbatim}
sudo apt install python3-pyqt6
pip install magboltz-gui
magboltz-gui
\end{verbatim}
\end{frame}
